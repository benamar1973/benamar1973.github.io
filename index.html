// server.js أو app.js const express = require('express'); const multer = require('multer'); const { v2: cloudinary } = require('cloudinary'); const { CloudinaryStorage } = require('multer-storage-cloudinary'); // تكوين Cloudinary من متغيرات البيئة cloudinary.config({ cloud_name: process.env.CLOUDINARY_CLOUD_NAME, api_key: process.env.CLOUDINARY_API_KEY, api_secret: process.env.CLOUDINARY_API_SECRET, }); // إعداد التخزين لـ multer const storage = new CloudinaryStorage({ cloudinary: cloudinary, params: { folder: 'marketplace_ads', allowed_formats: ['jpg', 'png', 'jpeg', 'mp4', 'mov'], resource_type: 'auto', // يكتشف تلقائيًا إذا كان صورة أو فيديو }, }); const upload = multer({ storage: storage }); const app = express(); app.use(express.json()); app.use(express.urlencoded({ extended: true })); // نموذج قاعدة بيانات (مثال باستخدام MongoDB) const Ad = require('./models/Ad'); // ستحتاج لإنشاء هذا النموذج // نقطة نهاية نشر الإعلان (استقبال النص والملفات) app.post('/api/ads', upload.array('media', 5), async (req, res) => { try { const { fullname, phone, description, contactMethod, contactValue } = req.body; // استخراج عناوين URL من الملفات المرفوعة const mediaUrls = req.files.map(file => file.path); // Cloudinary يعيد المسار الآمن const newAd = new Ad({ fullname, phone, description, contactMethod, contactValue, mediaUrls, createdAt: new Date() }); await newAd.save(); res.status(201).json({ message: 'تم النشر بنجاح', ad: newAd }); } catch (error) { console.error(error); res.status(500).json({ error: 'فشل النشر' }); } }); // نقطة نهاية جلب الإعلانات app.get('/api/ads', async (req, res) => { try { const ads = await Ad.find().sort({ createdAt: -1 }); res.json(ads); } catch (error) { res.status(500).json([]); } }); app.listen(5000, () => console.log('Server running on port 5000'));
